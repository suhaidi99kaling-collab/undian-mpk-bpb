Sambungan Terus ke Google Sheets

Sambungkan Google Sheets di bawah supaya setiap pencalonan dari telefon penduduk kampung disimpan secara kekal dan carta undian dikemas kini secara langsung (setiap 10 saat).

Kod Google Apps Script (Dua Hala & Undian Serentak)
  1. Buka Google Sheets baharu, klik menu Extensions > Apps Script.
  2. Padam kod sedia ada, tampal kod di bawah, kemudian klik ikon Save.
  3. Klik Deploy > New deployment > Web app.
  4. Tetapkan Execute as: Me, dan Who has access: Anyone.
  5. Klik Deploy, benarkan akses, kemudian salin Web app URL ke ruang di atas.
function getOrInitSheet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Tarikh & Masa", "Jawatan", "Nama Dicalonkan", "Nama Pencalon", "Catatan"]);
  }
  return sheet;
}

// Membaca semua rekod secara langsung untuk paparan carta undian
function doGet(e) {
  var sheet = getOrInitSheet();
  var values = sheet.getDataRange().getValues();
  var records = [];
  for (var i = values.length - 1; i >= 1; i--) {
    var row = values[i];
    if (!row[2] && !row[3]) continue;
    records.push({
      id: String(row[0] || ("row_" + i)),
      masa: String(row[1] || ""),
      jawatan: String(row[2] || ""),
      calon: String(row[3] || ""),
      pencalon: String(row[4] || ""),
      catatan: String(row[5] || "-")
    });
  }
  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok", records: records }))
    .setMimeType(ContentService.MimeType.JSON);
}

// Menyimpan atau memadam rekod dengan LockService (selamat untuk undian serentak)
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var sheet = getOrInitSheet();
    var payload = JSON.parse(e.postData.contents);

    if (payload.action === "delete" && payload.id) {
      var data = sheet.getDataRange().getValues();
      for (var i = data.length - 1; i >= 1; i--) {
        if (String(data[i][0]) === String(payload.id)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService
        .createTextOutput(JSON.stringify({ status: "deleted" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var items = payload.records ? payload.records : [payload];
    items.forEach(function(item) {
      sheet.appendRow([
        item.id || ("rec_" + new Date().getTime()),
        item.masa || "",
        item.jawatan || "",
        item.calon || "",
        item.pencalon || "",
        item.catatan || "-"
      ]);
    });

    return ContentService
      .createTextOutput(JSON.stringify({ status: "ok", count: items.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
